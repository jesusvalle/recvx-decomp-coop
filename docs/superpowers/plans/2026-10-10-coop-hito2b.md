# Cooperativo, hito 2b: P2 dispara — plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** P2 apunta, cambia de blanco y dispara con el mando 2, con la misma clase de arma que P1 y su munición, sin alterar a P1. Las vibraciones de P2 van al mando 2.

**Architecture:** El mando de P2 deja pasar los botones de combate (máscara `0x5FF`). `coopBegin`/`coopEnd` protegen además el puerto de vibración, el objeto de arma, el bit de arma vacía, los impactos del frame y la luz del fogonazo. La vibración del mando 2 es una tabla propia en el bloque COOP de ps2_sg_pdvib.c, a la que se desvía el puerto lógico 8.

**Tech Stack:** C (MWCC, estilo C89), libpad del SDK 2.0, Python 3.13 y PowerShell para las herramientas.

**Spec:** [docs/superpowers/specs/2026-10-10-coop-hito2b-design.md](../specs/2026-10-10-coop-hito2b-design.md)

## Global Constraints

## Review Focus

1. **Vibración pegada en el mando 2:** pausa, inventario, demo, P2 oculto o mando 2 desconectado con una vibración en curso. Esperado: el motor se para en menos de 10 frames o al momento. (Tarea 2: revisión de código de `coopVibStop2` en la salida de `coopSetPad2` y en `pdVibMxStop`.)
2. **P1 y P2 disparan a la vez:** dos impactos en el mismo enemigo y el mismo frame cuentan los dos; el arma de P1 no se anima por los disparos de P2 ni al revés. (Tarea 1: revisión del intercambio en `coopBegin`/`coopEnd`.)
3. **P2 vacía el arma compartida:** P1 ve el arma vacía y recarga igual que si la hubiera vaciado él. (Tarea 1: `gm_flg 0x40000` en `coopEnd`; Tarea 3: `ramread` de la munición.)
4. **P1 cambia de arma mientras P2 apunta:** P2 vuelve a reposo con la nueva arma, sin animaciones de la anterior. (Tarea 1: `coopStandP2` en `coopCloneWeapon`.)
5. **Armas con mira:** P2 no entra en la mira ni cambia la cámara de P1. (Tarea 1: máscara sin `0x10` con `WpnTab.flg & 0x20`.)

/* El fogonazo de P2 se enlaza a plp (lkflg 1), que bhControlLight resuelve con P1: se fija en la mano de P2. */
static void coopFlashToP2(void)
{
    LGT_WORK* lp;
    unsigned char* a;
    unsigned char* b;
    int i;
    int changed;

    if (rom->lgtp == NULL)
    {
        return;
    }

    lp = rom->lgtp;

    a = (unsigned char*)lp;
    b = (unsigned char*)&coop_lgt0;

    changed = 0;

    for (i = 0; i < (int)sizeof(LGT_WORK); i++)
    {
        if (a[i] != b[i])
        {
            changed = 1;
            break;
        }
    }

    if ((changed == 0) || (lp->lkflg != 1))
    {
        return;
    }

    if (lp->lkono == 0)
    {
        njCalcPoint(ply2.mtx, (NJS_POINT3*)&lp->lx, (NJS_POINT3*)&lp->px);
    }
    else
    {
        njCalcPoint(&ply2.mlwP->owP[lp->lkono].mtx, (NJS_POINT3*)&lp->lx, (NJS_POINT3*)&lp->px);
    }

    lp->lkflg = 0;
}

```c
#ifdef COOP

/* Vibración del mando 2 (P2): copia de la de P1 sobre su propia tabla, enviada al puerto 1. */
static PAD_ACT Pad_act2[20];
static unsigned char Ps2_pad_motor2[6] __attribute__((aligned(16)));
static int coop_vib_on;

Sint32 coopVibIsReady2(void)
{
    unsigned int pad_info;
    unsigned int pad_stat;
    unsigned char i;

    pad_stat = scePadGetState(1, 0);

    if ((pad_stat != scePadStateStable) && (pad_stat != scePadStateFindCTP1))
    {
        return 0;
    }

    i = 6;

    while (i--)
    {
        Ps2_pad_motor2[i] = 0;
    }

    pad_info = scePadInfoMode(1, 0, InfoModeCurExID, 0);

    if ((pad_info != 7) && (pad_info != 4))
    {
        return 0;
    }

    return 1;
}

Sint32 coopVibStart2(const PDS_VIBPARAM* param)
{
    unsigned char cnt;
    PAD_ACT* pPact;

    pPact = Pad_act2;
    cnt = 0;

    while (pPact->be_flag != 0)
    {
        cnt++;
        pPact++;

        if (cnt >= 20)
        {
            return -1;
        }
    }

    pPact->be_flag = 3;

    if (param->power > 85)
    {
        pPact->level = pPact->start = 255;
    }
    else
    {
        pPact->level = pPact->start = param->power + 170;
    }

    if (param->flag == 1)
    {
        pPact->time = 9;
    }
    else if (param->flag == 8)
    {
        pPact->time = 6;
    }
    else
    {
        pPact->time = 4;
    }

    return 0;
}

void coopVibStop2(void)
{
    int i;

    for (i = 0; i < 20; i++)
    {
        Pad_act2[i].be_flag = 0;
    }

    if (coop_vib_on != 0)
    {
        Ps2_pad_motor2[0] = 0;
        Ps2_pad_motor2[1] = 0;

        scePadSetActDirect(1, 0, Ps2_pad_motor2);

        coop_vib_on = 0;
    }
}

static void coopPadActStart2(PAD_ACT* pPact)
{
    unsigned char temp;

    if ((pPact->be_flag & 0x2))
    {
        if (pPact->f_add != 0)
        {
            pPact->level = (((short)pPact->start * 16) - (short)pPact->f_add) / 16;
        }

        temp = pPact->level;

        if (Ps2_pad_motor2[1] < temp)
        {
            Ps2_pad_motor2[1] = temp;
        }

        if (pPact->f_level != 0)
        {
            pPact->f_add += pPact->f_level;
        }
    }

    if (pPact->f_level != 0)
    {
        pPact->f_add += pPact->f_level;
    }
}

void coopPadActuater2(void)
{
    PAD_ACT* pPact;
    unsigned int cnt;

    pPact = Pad_act2;

    Ps2_pad_motor2[0] = 0;
    Ps2_pad_motor2[1] = 0;

    for (cnt = 0; cnt < 19; cnt++)
    {
        if ((pPact->be_flag & 0x3))
        {
            if (pPact->delay > 0)
            {
                pPact->delay--;
            }
            else if (pPact->time > 0)
            {
                coopPadActStart2(pPact);
                pPact->time--;
            }
            else
            {
                pPact->be_flag = 0;
            }
        }

        pPact++;
    }

    coop_vib_on = (Ps2_pad_motor2[1] != 0) ? 1 : 0;

    scePadSetActDirect(1, 0, Ps2_pad_motor2);
}

#endif
```

(La doble suma de `f_add` copia la de `Ps2_pad_act_start`.)
